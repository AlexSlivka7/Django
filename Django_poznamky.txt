3. nastavenie slovenciny pre projekt (nazov_projektu/settings.py):

	LANGUAGE_CODE = 'sk'

4. spustenie vývojového servera:
	python.exe .\manage.py runserver
	bezi na adrese http://127.0.0.1:8000

5. vytvorenie aplikacie:
	python.exe .\manage.py startapp (nazov_aplikacie)

6. registracia aplikacie do projektu (nazov_projektu/settings.py):
	INSTALLED_APPS = [
	    'django.contrib.admin',
	    'django.contrib.auth',
	    'django.contrib.contenttypes',
	    'django.contrib.sessions',
	    'django.contrib.messages',
	    'django.contrib.staticfiles',
 	    'nazov_aplikacie',
	]

8. vytvorenie urlpattern
8.1 nazov_projektu/urls.py
	from django.contrib import admin
	from django.urls import path, include

	urlpatterns = [
    	path('admin/', admin.site.urls),
    	path('skola/', include('nazov_aplikacie.urls')),
	]

8.2 vytvoriť súbor urls.py v priečinku aplikacie

	from django.urls import path
	from . import views

	urlpatterns = [
		path('', views.index, name='index'), # views.index = index je názov funkcie vo views.py
	]
